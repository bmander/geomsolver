cmake_minimum_required(VERSION 3.16)
project(solvent_libfive_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EIGEN REQUIRED eigen3)
find_path(BOOST_HEADERS boost/version.hpp HINTS "${BOOST_ROOT}")
find_library(LIBFIVE_LIBRARY NAMES five libfive PATHS "${LIBFIVE_BUILD}/libfive/src" NO_DEFAULT_PATH REQUIRED)
add_executable(solvent-libfive-bench main.cpp fixtures.cpp)
target_include_directories(solvent-libfive-bench PRIVATE "${LIBFIVE_SOURCE}/libfive/include" ${EIGEN_INCLUDE_DIRS} ${BOOST_HEADERS})
target_link_libraries(solvent-libfive-bench PRIVATE "${LIBFIVE_LIBRARY}")
